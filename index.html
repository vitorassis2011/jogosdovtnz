# Block Blast - Jogo estilo SameGame em Python com Pygame # Autor: ChatGPT (adaptado para fins educativos) import pygame import random import sys # ----------------------------- # Configurações do jogo (ajuste se desejar) # ----------------------------- COLS = 10 # colunas ROWS = 12 # linhas CELL = 40 # tamanho de cada bloco (em pixels) MARGIN = 4 # espaçamento entre blocos (em pixels) ORIGIN_X = 20 # posição X de origem da grade ORIGIN_Y = 60 # posição Y de origem da grade # Cores dos blocos (apenas tons para variedade) COLORS = [ (239, 83, 80), # vermelho (66, 165, 245), # azul (76, 175, 80), # verde (255, 235, 59), # amarelo (156, 39, 176), # roxo (255, 152, 0), # laranja (0, 0, 0), # preto (arredondado se quiser excluir) ] BG = (30, 30, 30) GRID_BORDER = (0, 0, 0) TEXT_COLOR = (255, 255, 255) HINT_COLOR = (255, 255, 255, 60) # Tamanho da janela calculado com base nas configurações acima WINDOW_WIDTH = ORIGIN_X * 2 + COLS * (CELL + MARGIN) WINDOW_HEIGHT = ORIGIN_Y + ROWS * (CELL + MARGIN) + 120 # espaço para HUD # ----------------------------- # Lógica do jogo # ----------------------------- class BlockBlastGame: def __init__(self, cols=COLS, rows=ROWS, colors=COLORS, cell_size=CELL, margin=MARGIN, origin=(ORIGIN_X, ORIGIN_Y)): self.cols = cols self.rows = rows self.cell_size = cell_size self.margin = margin self.origin_x, self.origin_y = origin self.colors = colors self.grid = [[None for _ in range(self.cols)] for _ in range(self.rows)] self.score = 0 self.best = 0 self.level = 1 self.game_over = False self.init_grid() # Inicializa grade com blocos aleatórios def init_grid(self): for y in range(self.rows): for x in range(self.cols): self.grid[y][x] = random.randrange(len(self.colors)) self.game_over = False self.score = 0 self.level = 1 def in_bounds(self, x, y): return 0 <= x < self.cols and 0 <= y < self.rows def get_color(self, x, y): return self.grid[y][x] def find_region(self, x, y): """Retorna um conjunto de coordenadas conectadas com a mesma cor.""" if not self.in_bounds(x, y): return set() color = self.grid[y][x] if color is None: return set() region = set() stack = [(x, y)] while stack: cx, cy = stack.pop() if (cx, cy) in region: continue if not self.in_bounds(cx, cy): continue if self.grid[cy][cx] != color: continue region.add((cx, cy)) for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]: nx, ny = cx + dx, cy + dy if self.in_bounds(nx, ny) and (nx, ny) not in region: stack.append((nx, ny)) return region def remove_region(self, region): for (x, y) in region: self.grid[y][x] = None self.score += len(region) ** 2 * self.level self.best = max(self.best, self.score) def apply_gravity(self): # Para cada coluna, faz os blocos caírem mantendo a ordem for x in range(self.cols): blocks = [] for y in range(self.rows - 1, -1, -1): if self.grid[y][x] is not None: blocks.append(self.grid[y][x]) y = self.rows - 1 for b in blocks: self.grid[y][x] = b y -= 1 for yy in range(y, -1, -1): self.grid[yy][x] = None def compress_columns(self): # Move colunas não vazias para a esquerda, mantendo ordem vertical write_x = 0 for x in range(self.cols): column = [self.grid[y][x] for y in range(self.rows)] if any(v is not None for v in column): for y in range(self.rows): self.grid[y][write_x] = column[y] write_x += 1 # Limpa colunas extras à direita for x in range(write_x, self.cols): for y in range(self.rows): self.grid[y][x] = None def fill_empty_cells(self): # Preenche espaços vazios no topo com novos blocos for x in range(self.cols): for y in range(self.rows): if self.grid[y][x] is None: self.grid[y][x] = random.randrange(len(self.colors)) def has_moves(self): # Verifica se ainda há moves disponíveis (regiões de tamanho >= 2) visited = set() for y in range(self.rows): for x in range(self.cols): if self.grid[y][x] is None: continue region = self.find_region(x, y) if len(region) >= 2: return True return False def click(self, gx, gy): """Processa clique na grade (gx, gy) em coordenadas da grade.""" if self.game_over: return if not (0 <= gx < self.cols and 0 <= gy < self.rows): return if self.grid[gy][gx] is None: return region = self.find_region(gx, gy) if len(region) >= 2: self.remove_region(region) self.apply_gravity() self.compress_columns() self.fill_empty_cells() # Atualiza nível em base ao score new_level = max(1, self.score // 100 + 1) self.level = new_level if not self.has_moves(): self.game_over = True def reset(self): self.grid = [[None for _ in range(self.cols)] for _ in range(self.rows)] self.init_grid() self.score = 0 self.best = 0 self.level = 1 self.game_over = False def grid_to_screen(self, gx, gy): """Retorna o retângulo da célula na tela (em pixels).""" x = self.origin_x + gx * (self.cell_size + self.margin) y = self.origin_y + gy * (self.cell_size + self.margin) return pygame.Rect(x, y, self.cell_size, self.cell_size) # Propriedades simples para facilitar o uso externo @property def origin(self): return (self.origin_x, self.origin_y) @property def origin_x(self): return self._origin_x @origin_x.setter def origin_x(self, v): self._origin_x = v @property def origin_y(self): return self._origin_y @origin_y.setter def origin_y(self, v): self._origin_y = v @property def cell_size(self): return self._cell_size @cell_size.setter def cell_size(self, v): self._cell_size = v @property def margin(self): return self._margin @margin.setter def margin(self, v): self._margin = v # ----------------------------- # Main (loop do jogo) # ----------------------------- def main(): pygame.init() pygame.display.set_caption("Block Blast - Jogo em Python") screen = pygame.display.set_mode((WINDOW_WIDTH, WINDOW_HEIGHT)) clock = pygame.time.Clock() font = pygame.font.SysFont(None, 28) big_font = pygame.font.SysFont(None, 48) game = BlockBlastGame() # Exibir FPS ao lado; opcional show_fps = False while True: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: mx, my = event.pos # Convertendo posição da tela para coordenadas da grade gx = (mx - ORIGIN_X) // (CELL + MARGIN) gy = (my - ORIGIN_Y) // (CELL + MARGIN) if 0 <= gx < COLS and 0 <= gy < ROWS: game.click(int(gx), int(gy)) if event.type == pygame.KEYDOWN: if event.key == pygame.K_r: game.reset() if event.key == pygame.K_h: show_fps = not show_fps # Desenho screen.fill(BG) # Desenhar grade de blocos for y in range(ROWS): for x in range(COLS): color_idx = game.grid[y][x] if color_idx is None: continue color = game.colors[color_idx] rect = pygame.Rect( ORIGIN_X + x * (CELL + MARGIN), ORIGIN_Y + y * (CELL + MARGIN), CELL, CELL ) # Bloco pygame.draw.rect(screen, color, rect) # Borda do bloco pygame.draw.rect(screen, GRID_BORDER, rect, 2) # HUD (Pontuação / Nível) score_surf = font.render(f"Pontos: {game.score}", True, TEXT_COLOR) level_surf = font.render(f"Nível: {game.level}", True, TEXT_COLOR) best_surf = font.render(f"Melhor: {game.best}", True, TEXT_COLOR) screen.blit(score_surf, (10, 10)) screen.blit(level_surf, (10, 38)) screen.blit(best_surf, (10, 66)) # Jogo encerrado if game.game_over: over_text = big_font.render("FIM DO JOGO!", True, (255, 0, 0)) sub_text = font.render("Pressione R para reiniciar", True, (200, 200, 200)) screen.blit(over_text, ( WINDOW_WIDTH // 2 - over_text.get_width() // 2, WINDOW_HEIGHT // 2 - over_text.get_height())) screen.blit(sub_text, ( WINDOW_WIDTH // 2 - sub_text.get_width() // 2, WINDOW_HEIGHT // 2 + 10)) # Instrução rápida instr = font.render("Clique em blocos de mesma cor (≥2). R para reset.", True, (180,180,180)) screen.blit(instr, (ORIGIN_X, WINDOW_HEIGHT - 40)) pygame.display.flip() clock.tick(60) if __name__ == "__main__": main()
